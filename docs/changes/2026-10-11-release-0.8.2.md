assert.equal(newestEntry.match(/version: '([^']+)'/)?.[1], packageDocument.version)
assert.match(newestEntry, /date: 'October 11, 2026'/)
assert.match(newestEntry, /WordPress 7\.1\.3/)
assert.match(newestEntry, /fallback/)

console.log('Release versions agree: 0.8.2; changelog date and WordPress fallback content verified.')
